### 3.2 핵심 파일 예시

**backend/src/services/PICalculationService.ts** (★ 계산 엔진)

```typescript
import { PILineItem } from '../types';

export class PICalculationService {
  /**
   * 판매가 계산 (이익률 기반)
   * 
   * 예: 10% 이익 → 판매가 = 매입가 / 0.9
   */
  calculateSalePrice(
    costKRW: number,
    exchangeRate: number,
    profitMargin: number
  ): number {
    const costUSD = costKRW / exchangeRate;
    const salePrice = costUSD / (1 - profitMargin);
    return Math.round(salePrice * 10000) / 10000;
  }

  /**
   * PI 라인 아이템 계산
   */
  calculateLineItem(item: PILineItem): PILineItem {
    item.costUSD = Math.round(
      (item.costKRW / item.exchangeRate) * 10000
    ) / 10000;

    item.salePriceUSD = this.calculateSalePrice(
      item.costKRW,
      item.exchangeRate,
      item.profitMargin
    );

    item.totalUSD = Math.round(
      item.salePriceUSD * item.quantity * 10000
    ) / 10000;

    return item;
  }

  /**
   * 전체 PI 합계 계산
   */
  calculatePI(
    lineItems: PILineItem[]
  ): { subtotal: number; grandTotal: number } {
    const subtotal = lineItems.reduce(
      (sum, item) => sum + item.totalUSD,
      0
    );

    return {
      subtotal: Math.round(subtotal * 10000) / 10000,
      grandTotal: Math.round(subtotal * 10000) / 10000
    };
  }
}
```

**backend/src/services/RevisionService.ts** (★ Revision 자동 생성)

```typescript
import { Database } from '../utils/db';

export class RevisionService {
  constructor(private db: Database) {}

  /**
   * 협상 합의 후 Revision 자동 생성
   */
  async onNegotiationAgreed(negotiation: {
    neg_id: string;
    supplier_id: string;
    product_id: string;
    final_agreed_cost_krw: number;
    supplier_name: string;
  }) {
    // Step 1: 영향받는 PI 찾기
    const affectedPIs = await this.db.query(`
      SELECT DISTINCT pi.pi_id, pi.pi_number, pi.current_version
      FROM proforma_invoices pi
      JOIN pi_revisions pr ON pi.pi_id = pr.pi_id 
        AND pr.version = pi.current_version
      JOIN pi_revision_line_items pril 
        ON pr.pr_id = pril.pi_revision_id
      WHERE pril.supplier_id = $1
        AND pril.product_id = $2
        AND pi.status IN ('draft', 'sent')
    `, [negotiation.supplier_id, negotiation.product_id]);

    const createdRevisions = [];

    // Step 2: 각 PI마다 새 Revision 생성
    for (const pi of affectedPIs) {
      const nextVersion = this.getNextVersion(pi.current_version);

      // Step 2.1: 최신 Revision 조회
      const latestRevision = await this.db.query(`
        SELECT * FROM pi_revisions 
        WHERE pi_id = $1 AND version = $2
      `, [pi.pi_id, pi.current_version]);

      // Step 2.2: 라인 아이템 조회
      const lineItems = await this.db.query(`
        SELECT * FROM pi_revision_line_items
        WHERE pi_revision_id = $1
      `, [latestRevision[0].pr_id]);

      // Step 2.3: 새 Revision 생성
      const newRevision = {
        pi_id: pi.pi_id,
        version: nextVersion,
        exchange_rate: latestRevision[0].exchange_rate,
        revision_reason: `${negotiation.supplier_name} 가격 합의`,
        related_negotiation_id: negotiation.neg_id,
        status: 'draft',
        created_by: 'system'
      };

      // Insert와 함께 반환
      const insertedRevision = await this.db.query(`
        INSERT INTO pi_revisions (...)
        VALUES (...)
        RETURNING *
      `);

      // Step 2.4: 라인 아이템 복사 및 수정
      for (const item of lineItems) {
        if (item.supplier_id === negotiation.supplier_id) {
          // 가격 변경 라인
          const newCostUSD = negotiation.final_agreed_cost_krw 
            / item.exchange_rate;
          const newSalePrice = newCostUSD / (1 - item.profit_margin);

          // cost_change_log 기록
          await this.db.query(`
            INSERT INTO cost_change_log (...)
            VALUES (...)
          `);

          // 새 라인 아이템 저장
          await this.db.query(`
            INSERT INTO pi_revision_line_items (...)
            VALUES (...)
          `);
        }
      }

      // PI의 current_version 업데이트
      await this.db.query(`
        UPDATE proforma_invoices 
        SET current_version = $1, updated_at = NOW()
        WHERE pi_id = $2
      `, [nextVersion, pi.pi_id]);

      createdRevisions.push({
        pi_number: pi.pi_number,
        new_version: nextVersion
      });
    }

    return createdRevisions;
  }

  private getNextVersion(current: string): string {
    const charCode = current.charCodeAt(0);
    if (charCode >= 65 && charCode < 90) {
      return String.fromCharCode(charCode + 1);
    }
    return 'A';
  }
}
```

**frontend/src/hooks/useCalculation.ts** (★ 계산 훅)

```typescript
import { useState, useCallback } from 'react';

interface CalculationInput {
  costKRW: number;
  exchangeRate: number;
  profitMargin: number;
}

interface CalculationOutput {
  costUSD: number;
  salePriceUSD: number;
}

export const useCalculation = () => {
  const calculateSalePrice = useCallback(
    (input: CalculationInput): CalculationOutput => {
      const { costKRW, exchangeRate, profitMargin } = input;

      // Step 1: KRW → USD
      const costUSD = Math.round(
        (costKRW / exchangeRate) * 10000
      ) / 10000;

      // Step 2: 판매가 계산 (이익률 기반)
      const salePriceUSD = Math.round(
        (costUSD / (1 - profitMargin)) * 10000
      ) / 10000;

      return { costUSD, salePriceUSD };
    },
    []
  );

  return { calculateSalePrice };
};
```

### 4.4 비즈니스 로직 구현 (Custom Actions)

**Calculation: 판매가 계산**

```
AntiGravity의 "Custom Action" 문법:

Action: CalculateSalePrice
Input:
  - costKRW: Number
  - exchangeRate: Number
  - profitMargin: Number

Logic:
  costUSD = costKRW / exchangeRate
  salePriceUSD = costUSD / (1 - profitMargin)
  
  if salePriceUSD > 0:
    return { costUSD, salePriceUSD }
  else:
    error "Invalid calculation"
```

**Workflow: Revision 자동 생성**

```
Trigger: supplier_negotiations.status = "agreed"

Actions:
  1. Query affected PIs
     SELECT * FROM proforma_invoices
     WHERE pi_id IN (
       SELECT pi_id FROM pi_revisions
       WHERE supplier_id = $supplier_id
     )
  
  2. For each PI:
     a) Calculate next version (A→B, B→C)
     b) Create new pi_revisions row
     c) Copy pi_revision_line_items
     d) Update costs for affected lines
     e) Insert cost_change_log
  
  3. Send Email
     To: sales@ysacc.co.kr
     Subject: "Revision created: Rev. X"
```

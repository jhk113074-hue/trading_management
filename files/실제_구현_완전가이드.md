**Day 3-5: 매입가 마스터 DB & 자동 저장**
```
Task 2.1.4: 매입가 마스터 테이블
  테이블: cost_master
    cm_id (UUID) - 기본키
    product_id (UUID FK) - 상품
    supplier_id (UUID FK) - 공급사
    po_id (UUID FK) - PO 참조
    po_line_item_id (UUID FK) - PO 라인 참조
    po_number (VARCHAR 100) - PO 번호
    unit_price (DECIMAL 12,2) - 단가 (KRW) ★
    currency (VARCHAR 3) - KRW
    unit (VARCHAR 50) - 단위
    effective_date (DATE) - PO 발주일
    is_latest (BOOLEAN) - 최신 여부 (TRUE/FALSE)
    version (INT) - 버전 (1, 2, 3...)
    notes (TEXT)
    created_at (TIMESTAMP)
    created_by (UUID FK)
  
  제약조건:
    - (product_id, supplier_id, version) UNIQUE

Task 2.1.5: PO 저장 시 매입가 자동 저장 로직
  흐름:
    1. PO 작성 완료 → [확정] 클릭
    2. 유효성 검사
       ├─ 공급사 선택 ✓
       ├─ 상품 선택 ✓
       ├─ 수량 > 0 ✓
       └─ 단가 > 0 ✓
    3. po_line_items 저장
    4. 각 라인마다:
       ├─ 기존 cost_master 조회 (is_latest = TRUE)
       ├─ 기존 있으면: is_latest = FALSE로 업데이트
       ├─ 새 cost_master 레코드 생성
       │  └─ unit_price = PO의 unit_price
       │  └─ is_latest = TRUE
       │  └─ po_number = PO-2026-001
       │  └─ effective_date = PO date
       └─ cost_master 저장
    5. PO 상태 = confirmed
    6. 성공 메시지 표시
```

-- 수익성 분석 뷰
CREATE VIEW v_profitability AS
SELECT 
  pi.pi_number,
  pi.customer_id,
  pril.product_id,
  pril.quantity,
  pril.cost_krw,
  pril.exchange_rate,
  
  -- 원가 계산
  (pril.cost_krw * pril.quantity) AS cost_krw_total,
  (pril.cost_krw * pril.quantity / pril.exchange_rate) AS cost_usd_total,
  
  -- 판매가 계산
  pril.sale_price_usd,
  pril.total_usd AS revenue_usd,
  
  -- 이익 계산
  (pril.total_usd - (pril.cost_krw * pril.quantity / pril.exchange_rate)) AS profit_usd,
  ROUND(
    (pril.total_usd - (pril.cost_krw * pril.quantity / pril.exchange_rate)) 
    / pril.total_usd * 100, 2
  ) AS profit_margin_percent
  
FROM proforma_invoices pi
JOIN pi_revisions pr ON pi.id = pr.pi_id AND pr.version = pi.current_version
JOIN pi_revision_line_items pril ON pr.id = pril.pi_revision_id;

-- 트리거: PO 번호 자동 생성
CREATE OR REPLACE FUNCTION generate_po_number()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.po_number IS NULL THEN
    NEW.po_number := 'PO-' || TO_CHAR(NEW.po_date, 'YYYY') || '-' 
                     || LPAD(nextval('po_number_seq')::text, 3, '0');
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_po_number
BEFORE INSERT ON purchase_orders
FOR EACH ROW
EXECUTE FUNCTION generate_po_number();

BEGIN
  IF NEW.pi_number IS NULL THEN
    NEW.pi_number := 'PI-TMS-' || TO_CHAR(NEW.pi_date, 'YYYY') || '-' 
                     || LPAD(nextval('pi_number_seq')::text, 2, '0');

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_pi_number
BEFORE INSERT ON proforma_invoices

## 5. 계산 엔진 구현

### 5.1 이익률 계산 (핵심)

```typescript
/**
 * 판매가 계산 (이익률 기반)
 * 
 * 예: 10% 이익을 원할 때
 * 판매가 = 매입가 / (1 - 0.10)
 */

function calculateSalePrice(
  costKRW: number,
  exchangeRate: number,
  profitMargin: number  // 0.10 (10%)
): number {
  // Step 1: KRW → USD 변환
  const costUSD = costKRW / exchangeRate;
  
  // Step 2: 이익률 적용
  const salePrice = costUSD / (1 - profitMargin);
  
  // Step 3: 4자리 반올림
  return Math.round(salePrice * 10000) / 10000;
}

// 사용 예
const costKRW = 5000;
const exchangeRate = 1468.96;
const profitMargin = 0.10;  // 10%

const salePrice = calculateSalePrice(costKRW, exchangeRate, profitMargin);
console.log(salePrice);  // 3.7778 USD

// 검증
const costUSD = costKRW / exchangeRate;  // 3.40 USD
const actualProfit = salePrice - costUSD;  // 0.3778 USD
const actualMargin = actualProfit / salePrice;  // 0.10 (10%) ✓
```

### 5.2 PI 라인 아이템 계산

```typescript
interface PILineItem {
  productId: string;
  productName: string;
  quantity: number;
  unit: string;
  costKRW: number;      // cost_master에서 로드
  exchangeRate: number; // PI의 환율
  profitMargin: number; // 0.10 (10%)
  
  // 계산 결과
  costUSD: number;
  salePriceUSD: number;
  totalUSD: number;
}

function calculateLineItem(item: PILineItem): PILineItem {
  // Step 1: 매입가 USD 계산
  item.costUSD = item.costKRW / item.exchangeRate;
  
  // Step 2: 판매가 USD 계산
  item.salePriceUSD = item.costUSD / (1 - item.profitMargin);
  
  // Step 3: 합계 USD 계산
  item.totalUSD = item.salePriceUSD * item.quantity;
  
  // Step 4: 반올림
  item.costUSD = Math.round(item.costUSD * 10000) / 10000;
  item.salePriceUSD = Math.round(item.salePriceUSD * 10000) / 10000;
  item.totalUSD = Math.round(item.totalUSD * 10000) / 10000;
  
  return item;
}
```

### 5.3 Revision 자동 생성 로직

```typescript
/**
 * 협상 합의 후 자동으로 새 Revision 생성
 */

async function onSupplierNegotiationAgreed(negotiation) {
  // Step 1: 이 공급처의 가격을 사용한 모든 PI 찾기
  const affectedPIs = await db.query(`
    SELECT DISTINCT pi.pi_id, pi.pi_number, pi.current_version
    FROM proforma_invoices pi
    JOIN pi_revisions pr ON pi.pi_id = pr.pi_id 
      AND pr.version = pi.current_version
    JOIN pi_revision_line_items pril ON pr.pr_id = pril.pi_revision_id
    WHERE pril.supplier_id = ?
      AND pril.product_id = ?
      AND pi.status IN ('draft', 'sent')
    ORDER BY pi.created_at DESC
  `, [negotiation.supplier_id, negotiation.product_id]);

  // Step 2: 각 PI마다 새 Revision 생성
  const createdRevisions = [];
  
  for (const pi of affectedPIs) {
    // Step 2.1: 다음 Revision 번호 계산
    const nextVersion = getNextVersion(pi.current_version);  // A→B, B→C
    
    // Step 2.2: 최신 Revision 조회
    const latestRevision = await db.query(`
      SELECT * FROM pi_revisions 
      WHERE pi_id = ? AND version = ?
    `, [pi.pi_id, pi.current_version]);
    
    // Step 2.3: 새 Revision 레코드 생성
    const newRevision = {
      pi_id: pi.pi_id,
      version: nextVersion,
      exchange_rate: latestRevision.exchange_rate,
      payment_terms: latestRevision.payment_terms,
      revision_reason: `${negotiation.supplier_name} 가격 합의 (${negotiation.current_cost_krw}→${negotiation.final_agreed_cost_krw} KRW)`,
      related_negotiation_id: negotiation.neg_id,
      status: 'draft',
      created_by: getCurrentUserId(),
      created_at: new Date()
    };
    
    // Step 2.4: pi_revision_line_items 복사 및 수정
    const lineItems = await db.query(`
      SELECT * FROM pi_revision_line_items
      WHERE pi_revision_id = ?
    `, [latestRevision.pr_id]);
    
    const newLineItems = lineItems.map(item => {
      if (item.supplier_id === negotiation.supplier_id) {
        // 이 라인은 가격이 변경됨
        const oldCostKRW = item.cost_krw;
        const newCostKRW = negotiation.final_agreed_cost_krw;
        
        // 새 판매가 계산
        const newCostUSD = newCostKRW / item.exchange_rate;
        const newSalePrice = newCostUSD / (1 - item.profit_margin);
        const newTotal = newSalePrice * item.quantity;
        
        // 변경 이력 기록
        const costChangeLog = {
          pi_id: pi.pi_id,
          pi_version: nextVersion,
          product_id: item.product_id,
          supplier_id: item.supplier_id,
          old_cost_krw: oldCostKRW,
          new_cost_krw: newCostKRW,
          change_percent: ((newCostKRW - oldCostKRW) / oldCostKRW * 100),
          old_sale_price_usd: item.sale_price_usd,
          new_sale_price_usd: newSalePrice,
          sale_price_impact: newSalePrice - item.sale_price_usd
        };
        
        await db.save('cost_change_log', costChangeLog);
        
        // 새 라인 아이템 생성
        return {
          ...item,
          cost_krw: newCostKRW,
          sale_price_usd: newSalePrice,
          total_usd: newTotal,
          change_reason: `${negotiation.supplier_name} 가격 변동`,
          previous_sale_price_usd: item.sale_price_usd,
          price_change_percent: ((newSalePrice - item.sale_price_usd) / item.sale_price_usd * 100)
        };
      } else {
        // 이 라인은 변경 없음
        return item;
      }
    });
    
    // Step 2.5: DB에 저장
    await db.save('pi_revisions', newRevision);
    
    // 새 라인 아이템 저장
    for (const lineItem of newLineItems) {
      await db.save('pi_revision_line_items', {
        ...lineItem,
        pi_revision_id: newRevision.pr_id
      });
    }
    
    // Step 2.6: PI의 current_version 업데이트
    await db.update('proforma_invoices',
      { pi_id: pi.pi_id },
      { current_version: nextVersion, updated_at: new Date() }
    );
    
    createdRevisions.push({
      pi_number: pi.pi_number,
      new_version: nextVersion,
      created_at: new Date()
    });
  }
  
  // Step 3: 결과 반환
  return {
    negotiation_id: negotiation.neg_id,
    status: 'agreed',
    created_revisions: createdRevisions,
    affected_pi_count: affectedPIs.length
  };
}

// 다음 버전 계산
function getNextVersion(currentVersion: string): string {
  const charCode = currentVersion.charCodeAt(0);
  if (charCode >= 65 && charCode < 90) {
    return String.fromCharCode(charCode + 1);
  }
  return 'A';  // Z 이후 A로
}
```
